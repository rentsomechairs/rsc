rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    /* Primary owner/admin account. */
    function isMe() {
      return request.auth != null &&
             request.auth.uid == "2OjzWe94F5RJrFdgx3fSF8gceuF2";
    }

    function isSignedIn() {
      return request.auth != null;
    }

    /* Employee / secondary-account helpers. */
    function employeeProfileExists() {
      return isSignedIn() &&
             exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    function currentProfile() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function isApprovedEmployee() {
      return employeeProfileExists() &&
             currentProfile().role == "employee" &&
             currentProfile().status == "approved";
    }

    function isApprovedSecondary() {
      return employeeProfileExists() &&
             currentProfile().role == "secondary" &&
             currentProfile().status == "approved" &&
             currentProfile().primaryEmployeeId is string;
    }

    function isApprovedWorker() {
      return isApprovedEmployee() || isApprovedSecondary();
    }

    /* For a secondary login, permissions follow the primary employee. */
    function effectiveEmployeeUid() {
      return isApprovedSecondary()
        ? currentProfile().primaryEmployeeId
        : request.auth.uid;
    }

    function isAssignedEmployee() {
      return isApprovedWorker() &&
             resource.data.assignedEmployeeId == effectiveEmployeeUid();
    }

    /*
     * Employee/approved secondary may make ONLY the limited order-progress
     * changes available in the employee UI.
     */
    function isEmployeeOrderProgressUpdate() {
      return isApprovedWorker() &&
             resource.data.assignedEmployeeId == effectiveEmployeeUid() &&
             request.resource.data.assignedEmployeeId == effectiveEmployeeUid() &&

             request.resource.data
               .diff(resource.data)
               .affectedKeys()
               .hasOnly([
                 "exchangeTime",
                 "returnTime",
                 "status",
                 "completedAt",
                 "paymentStatus",
                 "amountPaid",
                 "depositPaidAmount",
                 "amountRemaining",
                 "updatedAt"
               ]) &&

             (
               request.resource.data.status == resource.data.status ||
               (
                 resource.data.status == "Confirmed" &&
                 request.resource.data.status == "In-Progress"
               ) ||
               (
                 resource.data.status == "Confirmed" &&
                 request.resource.data.status == "Completed"
               ) ||
               (
                 resource.data.status == "In-Progress" &&
                 request.resource.data.status == "Completed"
               )
             ) &&

             (
               request.resource.data.status != "Completed" ||
               resource.data.paymentStatus == "Free" ||
               request.resource.data.paymentStatus == "Free" ||
               (
                 request.resource.data.paymentStatus == "Paid" &&
                 request.resource.data.amountRemaining == 0
               )
             );
    }

    /* Employee/secondary may synchronize the same limited tracking fields. */
    function isEmployeeTrackingProgressUpdate(orderId) {
      return isApprovedWorker() &&
             exists(/databases/$(database)/documents/orders/$(orderId)) &&
             get(/databases/$(database)/documents/orders/$(orderId)).data.assignedEmployeeId == effectiveEmployeeUid() &&
             request.resource.data
               .diff(resource.data)
               .affectedKeys()
               .hasOnly([
                 "exchangeTime",
                 "returnTime",
                 "status",
                 "paymentStatus",
                 "amountPaid",
                 "depositPaidAmount",
                 "amountRemaining",
                 "updatedAt"
               ]);
    }

    /* Quick Picker order helpers. */
    function isQuickPickerOrder() {
      return request.resource.data.source == "quick-picker" &&
             request.resource.data.status == "Pending";
    }

    function isExistingQuickPickerOrder() {
      return resource.data.source == "quick-picker" &&
             request.resource.data.source == "quick-picker" &&
             request.resource.data.status == "Pending";
    }

    /* Primary employee signup. */
    function isValidEmployeeSignup(uid) {
      return isSignedIn() &&
             request.auth.uid == uid &&
             request.resource.data.uid == uid &&
             request.resource.data.role == "employee" &&
             request.resource.data.status == "pending" &&
             request.resource.data.email == request.auth.token.email &&
             request.resource.data.firstName is string &&
             request.resource.data.lastName is string &&
             request.resource.data.phone is string &&
             request.resource.data.pickupAddress is string;
    }

    /* Secondary login created from an employee's personal invite link. */
    function isValidSecondarySignup(uid) {
      return isSignedIn() &&
             request.auth.uid == uid &&
             request.resource.data.uid == uid &&
             request.resource.data.role == "secondary" &&
             request.resource.data.status == "pending_primary" &&
             request.resource.data.email == request.auth.token.email &&
             request.resource.data.primaryEmployeeId is string &&
             request.resource.data.primaryEmployeeId.size() > 0 &&
             request.resource.data.firstName is string &&
             request.resource.data.lastName is string &&
             request.resource.data.phone is string;
    }

    /* Preserve your existing employee self-edit behavior without allowing
       employees to approve themselves, change role, UID, or email. */
    function employeeKeepsProtectedFields() {
      return request.resource.data.uid == resource.data.uid &&
             request.resource.data.role == resource.data.role &&
             request.resource.data.status == resource.data.status &&
             request.resource.data.email == resource.data.email;
    }

    /* Primary employee may approve/revoke only secondary accounts attached
       to that primary employee. */
    function isPrimarySecondaryApproval() {
      return isApprovedEmployee() &&
             resource.data.role == "secondary" &&
             resource.data.primaryEmployeeId == request.auth.uid &&
             request.resource.data.uid == resource.data.uid &&
             request.resource.data.email == resource.data.email &&
             request.resource.data.role == resource.data.role &&
             request.resource.data.primaryEmployeeId == resource.data.primaryEmployeeId &&
             request.resource.data
               .diff(resource.data)
               .affectedKeys()
               .hasOnly(["status", "approvedByPrimaryAt", "updatedAt"]) &&
             request.resource.data.status in ["approved", "pending_primary"];
    }

    /* Employee and secondary profiles. */
    match /users/{uid} {
      allow get: if isMe() ||
                   (isSignedIn() && request.auth.uid == uid) ||
                   (isApprovedEmployee() && resource.data.role == "secondary" && resource.data.primaryEmployeeId == request.auth.uid) ||
                   (isApprovedSecondary() && uid == effectiveEmployeeUid());

      /* Primary employees may query only their linked secondary accounts. */
      allow list: if isMe() ||
                    (isApprovedEmployee() &&
                     resource.data.role == "secondary" &&
                     resource.data.primaryEmployeeId == request.auth.uid);

      allow create: if isValidEmployeeSignup(uid) || isValidSecondarySignup(uid);

      allow update: if isMe() ||
                       (
                         isSignedIn() &&
                         request.auth.uid == uid &&
                         resource.data.role == "employee" &&
                         employeeKeepsProtectedFields()
                       ) ||
                       isPrimarySecondaryApproval();

      allow delete: if isMe();
    }

    /* Weekly schedule + date-specific changes.
       Secondary logins read/write the PRIMARY employee's schedule. */
    match /schedules/{uid} {
      allow get: if isMe() ||
                   (isApprovedEmployee() && request.auth.uid == uid) ||
                   (isApprovedSecondary() && effectiveEmployeeUid() == uid);

      allow list: if isMe();

      allow create, update: if isMe() ||
                               (isApprovedEmployee() && request.auth.uid == uid) ||
                               (isApprovedSecondary() && effectiveEmployeeUid() == uid);

      allow delete: if isMe();
    }

    /* Optional admin bootstrap/security document. */
    match /appSecurity/{docId} {
      allow read, write: if isMe();
    }

    match /settings/{docId} {
      allow read: if true;
      allow write: if isMe();
    }

    match /inventory/{docId} {
      allow read: if true;
      allow write: if isMe();
    }

    match /equipment/{docId} {
      allow read: if true;
      allow write: if isMe();
    }

    match /categories/{docId} {
      allow read: if true;
      allow write: if isMe();
    }

    /* Orders */
    match /orders/{orderId} {
      allow get, list: if request.auth == null || isMe() || isAssignedEmployee();

      allow create: if isMe() ||
                       (request.auth == null && isQuickPickerOrder());

      allow update: if isMe() ||
                       isEmployeeOrderProgressUpdate() ||
                       (request.auth == null && isExistingQuickPickerOrder());

      allow delete: if isMe();
    }

    /* Public/customer tracking. */
    match /orderTracking/{trackingCode} {
      allow read: if true;

      allow create: if isMe() ||
                       (request.auth == null && request.resource.data.status == "Pending");

      allow update: if isMe() ||
                       isEmployeeTrackingProgressUpdate(trackingCode) ||
                       (
                         request.auth == null &&
                         resource.data.status == "Pending" &&
                         request.resource.data.status == "Pending"
                       );

      allow delete: if isMe();
    }


    /* Employee payout requests.
     * - Owner/admin can read, create, mark paid, decline, or delete.
     * - An approved PRIMARY employee can create and read only their own requests.
     * - Secondary logins cannot request payouts.
     * - Employees cannot change status after submission.
     */
    match /payoutRequests/{requestId} {
      allow get, list: if isMe() ||
                       (
                         isApprovedEmployee() &&
                         resource.data.employeeUid == request.auth.uid
                       );

      allow create: if isMe() ||
                    (
                      isApprovedEmployee() &&
                      request.resource.data.employeeUid == request.auth.uid &&
                      request.resource.data.status == "pending" &&
                      request.resource.data.amount is number &&
                      request.resource.data.amount > 0 &&
                      request.resource.data.payoutAccountId is string &&
                      request.resource.data.payoutAccountNickname is string &&
                      request.resource.data.payoutMethod is string &&
                      request.resource.data.payoutDetail is string &&
                      request.resource.data.createdAt is string &&
                      request.resource.data.updatedAt is string
                    );

      allow update, delete: if isMe();
    }

    match /reviews/{reviewId} {
      allow read: if true;
      allow create: if true;
      allow update, delete: if isMe();
    }

    match /inquiries/{docId} {
      allow create: if true;
      allow read, update, delete: if isMe();
    }

    match /customers/{docId} {
      allow read, write: if isMe();
    }

    match /costs/{docId} {
      allow read, write: if isMe();
    }

    match /orderAuditLog/{docId} {
      allow create: if isMe() ||
                      (
                        request.auth == null &&
                        request.resource.data.actor in [
                          "quick-picker",
                          "quick-picker-notify"
                        ]
                      );
      allow read, update, delete: if isMe();
    }

    match /orderSnapshots/{docId} {
      allow read, write: if isMe();
    }

    /* Finance records remain owner-only. */
    match /financeExpenses/{docId} {
      allow read, write: if isMe();
    }

    match /financeIncome/{docId} {
      allow read, write: if isMe();
    }

    match /financeMileage/{docId} {
      allow read, write: if isMe();
    }

    match /financeVehicles/{docId} {
      allow read, write: if isMe();
    }

    match /financeAssets/{docId} {
      allow read, write: if isMe();
    }

    match /financeIncidents/{docId} {
      allow read, write: if isMe();
    }

    match /financeHomeOffice/{docId} {
      allow read, write: if isMe();
    }

    match /financeCategories/{docId} {
      allow read, write: if isMe();
    }

    match /financeMileageRates/{docId} {
      allow read, write: if isMe();
    }

    match /financeAttachments/{docId} {
      allow read, write: if isMe();
    }

    match /financeAudit/{docId} {
      allow read, create: if isMe();
      allow update, delete: if false;
    }

    /* Everything else remains owner-only. */
    match /{document=**} {
      allow read, write: if isMe();
    }
  }
}
